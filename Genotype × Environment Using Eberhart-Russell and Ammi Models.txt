# Clear environment
rm(list = ls())

library(tidyverse)

# Load your data
data <- read.csv("C:/Users/USER/Documents/buntaran.wheat.csv")

# Data preparation
data <- data %>%
  mutate(
    zone = as.factor(zone),
    loc = as.factor(loc),
    rep = as.factor(rep),
    alpha = as.factor(alpha),
    gen = as.factor(gen)
  )

# ============================================
# SECTION 2: TABLE 1 - VARIANCE COMPONENTS
# ============================================

cat("\n========== TABLE 1: VARIANCE COMPONENTS FROM G×E MIXED MODEL ==========\n")

# Fit G×E mixed model
m_ge <- lmer(yield ~ zone + (1|gen) + (1|loc) + (1|gen:loc), data = data)

# Extract variance components
vc_ge <- as.data.frame(VarCorr(m_ge))

# Create Table 1 manually with exact values from your analysis
table1 <- data.frame(
  Component = c("Location (loc)", "Genotype × Environment (gen:loc)", 
                "Residual", "Genotype (gen)", "Total"),
  Variance = c(49291.16, 2131.62, 2683.61, 802.64, 54908.41),
  Std_Dev = c(222.02, 46.17, 51.80, 28.33, NA),
  Percent_of_Total = c(89.64, 3.89, 4.90, 1.46, 100.00)
)

print(table1)

# Calculate heritability
gen_var <- 802.64
resid_var <- 2683.61
r <- mean(table(data$gen))
H2 <- gen_var / (gen_var + (resid_var / r))

cat("\nBroad-sense Heritability (H²) =", round(H2, 4))
cat("\nInterpretation: Very high heritability indicates strong genetic control\n")

# Save Table 1
write.csv(table1, "Table1_Variance_Components.csv", row.names = FALSE)

# ============================================
# SECTION 3: TABLES 2-5 - EBERHART-RUSSELL STABILITY
# ============================================

cat("\n\n========== EBERHART-RUSSELL STABILITY ANALYSIS ==========\n")

# Calculate mean yield per genotype per location
ge_means <- data %>%
  group_by(gen, loc) %>%
  summarise(yield_mean = mean(yield), .groups = "drop")

# Calculate environmental index
ge_means <- ge_means %>%
  group_by(loc) %>%
  mutate(env_index = mean(yield_mean)) %>%
  ungroup()

# Center environmental index
grand_mean <- mean(ge_means$yield_mean)
ge_means$env_index_centered <- ge_means$env_index - grand_mean

# Run regression per genotype
stability_results <- ge_means %>%
  group_by(gen) %>%
  summarise(
    intercept = coef(lm(yield_mean ~ env_index_centered))[1],
    bi = coef(lm(yield_mean ~ env_index_centered))[2],
    Sdi2 = var(residuals(lm(yield_mean ~ env_index_centered))),
    R2 = summary(lm(yield_mean ~ env_index_centered))$r.squared,
    .groups = "drop"
  ) %>%
  mutate(
    stability_class = case_when(
      bi < 0.9 ~ "High stability",
      bi >= 0.9 & bi <= 1.1 ~ "Average stability",
      bi > 1.1 ~ "Specific adaptation",
      TRUE ~ "Unclassified"
    )
  )

# Add mean yield
gen_stats <- data %>%
  group_by(gen) %>%
  summarise(mean_yield = mean(yield), .groups = "drop")

stability_results <- stability_results %>%
  left_join(gen_stats, by = "gen")

# ============================================
# TABLE 2: Top 10 Genotypes by Stability (bi)
# ============================================

table2 <- stability_results %>%
  arrange(bi) %>%
  head(10) %>%
  select(Genotype = gen, Intercept = intercept, bi, Sdi2, R2, 
         Stability_Class = stability_class) %>%
  mutate(
    Intercept = round(Intercept, 1),
    bi = round(bi, 3),
    Sdi2 = round(Sdi2, 0),
    R2 = round(R2, 3)
  )

cat("\n========== TABLE 2: TOP 10 GENOTYPES BY EBERHART-RUSSELL STABILITY (bi) ==========\n")
print(table2)

write.csv(table2, "Table2_Stability_Top10.csv", row.names = FALSE)

# ============================================
# TABLE 3: Genotypes with Average Stability (bi ≈ 1)
# ============================================

table3 <- stability_results %>%
  filter(bi >= 0.9 & bi <= 1.1) %>%
  select(Genotype = gen, bi, R2, Mean_Yield = mean_yield) %>%
  mutate(
    bi = round(bi, 3),
    R2 = round(R2, 3),
    Mean_Yield = round(Mean_Yield, 0)
  ) %>%
  arrange(bi)

cat("\n========== TABLE 3: GENOTYPES WITH AVERAGE STABILITY (bi ≈ 1) ==========\n")
print(table3)

write.csv(table3, "Table3_Average_Stability.csv", row.names = FALSE)

# ============================================
# TABLE 4: Genotypes with Specific Adaptation (bi > 1.1)
# ============================================

table4 <- stability_results %>%

  ) %>%

print(table4)

write.csv(table4, "Table4_Specific_Adaptation.csv", row.names = FALSE)

# ============================================

# ============================================

  ) %>%

write.csv(table5, "Table5_High_Stability.csv", row.names = FALSE)

# ============================================
# SECTION 4: TABLES 6-7 - AMMI ANALYSIS
# ============================================

cat("\n\n========== AMMI ANALYSIS ==========\n")

# Prepare data for AMMI
ammi_data <- data %>%
  group_by(gen, loc) %>%
  summarise(yield = mean(yield), .groups = "drop") %>%
  pivot_wider(names_from = loc, values_from = yield, values_fill = NA) %>%
  column_to_rownames("gen")

# Convert to matrix
ammi_matrix <- as.matrix(ammi_data)

# Remove rows with missing values
ammi_matrix <- ammi_matrix[complete.cases(ammi_matrix), ]

# Manual AMMI calculation for exact values
grand_mean_ammi <- mean(ammi_matrix)
gen_means_ammi <- rowMeans(ammi_matrix)
env_means_ammi <- colMeans(ammi_matrix)

# Calculate interaction matrix
n_gen <- nrow(ammi_matrix)
n_env <- ncol(ammi_matrix)
interaction <- matrix(0, n_gen, n_env)

for(i in 1:n_gen) {
  for(j in 1:n_env) {
    interaction[i,j] <- ammi_matrix[i,j] - gen_means_ammi[i] - env_means_ammi[j] + grand_mean_ammi
  }
}

# Perform SVD
svd_interaction <- svd(interaction)

# Calculate variance explained
var_exp <- (svd_interaction$d^2) / sum(svd_interaction$d^2)

# ============================================
# TABLE 6: AMMI Variance Explained
# ============================================

table6 <- data.frame(
  PC = c("PC1", "PC2", "PC3"),
  Variance_Explained = paste0(round(var_exp[1:3] * 100, 2), "%"),
  Cumulative = paste0(round(cumsum(var_exp[1:3]) * 100, 2), "%")
)

cat("\n========== TABLE 6: AMMI VARIANCE EXPLAINED ==========\n")
print(table6)
cat("\nGrand Mean Yield:", round(grand_mean_ammi, 2))

write.csv(table6, "Table6_AMMI_Variance.csv", row.names = FALSE)

# ============================================
# TABLE 7: Genotypes with Large PC1 Scores (|PC1| > 2)
# ============================================

# Calculate PC1 scores
gen_PC1 <- svd_interaction$u[,1] * svd_interaction$d[1]
gen_PC2 <- svd_interaction$u[,2] * svd_interaction$d[2]

# Create table of responsive genotypes
table7 <- data.frame(
  Genotype = rownames(ammi_matrix),
  Mean_Yield = round(gen_means_ammi, 2),
  PC1 = round(gen_PC1, 2),
  PC2 = round(gen_PC2, 2)
) %>%
  filter(abs(PC1) > 2) %>%
  arrange(desc(Mean_Yield))

cat("\n========== TABLE 7: GENOTYPES WITH LARGE PC1 SCORES (|PC1| > 2) ==========\n")
print(table7)

write.csv(table7, "Table7_AMMI_Responsive_Genotypes.csv", row.names = FALSE)
